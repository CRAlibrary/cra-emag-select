  <script>

      const API_URL = "https://script.google.com/macros/s/AKfycbxWqJMu2kgMFqs_7x-RfjUyRuO1Y86ZQJQUeCEqoCv4NKtBDejG3XN7n0z0dLnGymBbAw/exec";


      // ฟังก์ชันกลับหน้าแรก

      function goToIndex() {

          window.location.href = "index.html";

      }


      function handleLogin(e) {

          e.preventDefault();

            

          const user = document.getElementById('username').value;

          const pass = document.getElementById('password').value;

          const btn = document.getElementById('loginBtn');

          const errorMsg = document.getElementById('errorMsg');

          const errorText = document.getElementById('errorText');

            

          // เปลี่ยนสถานะปุ่ม

          btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> กำลังตรวจสอบ...';

          btn.disabled = true;

          btn.classList.add('opacity-70', 'cursor-not-allowed');

          errorMsg.classList.add('hidden');


          // เรียกใช้ API เพื่อยืนยันตัวตน

          fetch(API_URL, {

              method: 'POST',

              headers: {

                  'Content-Type': 'text/plain;charset=utf-8', 

              },

              body: JSON.stringify({

                  action: 'authenticateAdmin',

                  username: user,

                  password: pass

              })

          })

          .then(res => res.json())

          .then(response => {

              if (response.success) {

                  // จำลองการเก็บ Session แอดมินไว้ใน LocalStorage

                  localStorage.setItem('cra_admin_token', 'authenticated_' + new Date().getTime());

                  // เปลี่ยนไปหน้า Dashboard

                  window.location.href = "dashboard.html";

              } else {

                  // กรณีรหัสผิด

                  errorText.innerText = response.message || 'ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง';

                  errorMsg.classList.remove('hidden');

                  resetButton();

              }

          })

          .catch(error => {

              errorText.innerText = 'เกิดข้อผิดพลาดในการเชื่อมต่อเซิร์ฟเวอร์';

              errorMsg.classList.remove('hidden');

              resetButton();

          });

      }


      function resetButton() {

          const btn = document.getElementById('loginBtn');

          btn.innerHTML = '<span>เข้าสู่ระบบ</span> <i class="fas fa-sign-in-alt"></i>';

          btn.disabled = false;

          btn.classList.remove('opacity-70', 'cursor-not-allowed');

      }
