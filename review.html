  <script>

      const API_URL = "https://script.google.com/macros/s/AKfycbxWqJMu2kgMFqs_7x-RfjUyRuO1Y86ZQJQUeCEqoCv4NKtBDejG3XN7n0z0dLnGymBbAw/exec";

      

      let cartData = [];

      let userData = {};


      window.onload = function() {

          loadData();

      };


      function loadData() {

          // โหลดข้อมูลผู้ใช้จาก LocalStorage

          const savedUser = localStorage.getItem('cra_user_info');

          if (savedUser) {

              userData = JSON.parse(savedUser);

              document.getElementById('userInfoDisplay').innerHTML = `

                  <p><span class="font-medium">ชื่อ-นามสกุล:</span> ${userData.name || '-'}</p>

                  <p><span class="font-medium">อีเมล:</span> ${userData.email || '-'}</p>

                  <p><span class="font-medium">ประเภทผู้ใช้งาน:</span> ${userData.userType || '-'}</p>

                  <p><span class="font-medium">หน่วยงาน/คณะ:</span> ${userData.department || '-'}</p>

              `;

          }


          // โหลดข้อมูลตะกร้าสินค้า

          const savedCart = localStorage.getItem('cra_selected_mags');

          if (savedCart) {

              cartData = JSON.parse(savedCart);

              document.getElementById('totalItems').innerText = cartData.length;

              

              const tbody = document.getElementById('cartItemsList');

              let total = 0;

              let html = '';


              cartData.forEach((mag, index) => {

                  total += mag.price || 0;

                  html += `

                      <tr class="hover:bg-gray-50">

                          <td class="px-4 py-3 text-sm text-gray-500">${index + 1}</td>

                          <td class="px-4 py-3 text-sm font-medium text-gray-800">${mag.title}</td>

                          <td class="px-4 py-3 text-sm text-gray-500">${mag.category}</td>

                          <td class="px-4 py-3 text-sm text-gray-800 text-right">${(mag.price || 0).toLocaleString('th-TH', {minimumFractionDigits: 2})}</td>

                      </tr>

                  `;

              });


              tbody.innerHTML = html;

              document.getElementById('totalPrice').innerText = total.toLocaleString('th-TH', {minimumFractionDigits: 2});

          }

      }


      function goBack() {

          window.location.href = 'select.html';

      }


      function submitData() {

          document.getElementById('loader').classList.remove('hidden');

          document.getElementById('loader').classList.add('flex');

          document.getElementById('submitBtn').disabled = true;


          const payload = {

              user: userData,

              items: cartData,

              totalItems: cartData.length,

              totalPrice: cartData.reduce((sum, item) => sum + (item.price || 0), 0)

          };


          // เรียกใช้ฟังก์ชันเซฟข้อมูลผ่าน fetch

          fetch(API_URL, {

              method: 'POST',

              headers: {

                  'Content-Type': 'text/plain;charset=utf-8', // ใช้ text/plain เพื่อเลี่ยง CORS Preflight Options

              },

              body: JSON.stringify({

                  action: 'saveSelection',

                  payload: payload

              })

          })

          .then(res => res.json())

          .then(data => {

              if (data.success) {

                  // ถ้าเซฟสำเร็จ ให้ล้างตะกร้าทิ้ง แล้วเปลี่ยนหน้า

                  localStorage.removeItem('cra_selected_mags');

                  window.location.href = 'success.html';

              } else {

                  throw new Error(data.message || 'บันทึกไม่สำเร็จ');

              }

          })

          .catch(error => {

              document.getElementById('loader').classList.add('hidden');

              document.getElementById('loader').classList.remove('flex');

              document.getElementById('submitBtn').disabled = false;

              alert('เกิดข้อผิดพลาดในการบันทึกข้อมูล: ' + error.message);

          });

      }
