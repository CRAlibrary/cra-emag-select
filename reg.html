              domainSelect.add(new Option('cra.ac.th', 'cra.ac.th'));

              domainSelect.classList.add('bg-gray-200', 'pointer-events-none');

          }

      }


      function handlePrefixChange() {

          const prefix = document.getElementById('prefix').value;

          const customWrapper = document.getElementById('custom-prefix-wrapper');

          const customInput = document.getElementById('customPrefix');

          const nameFields = document.getElementById('name-fields');


          if (prefix === 'อื่นๆ') {

              customWrapper.classList.remove('hidden');

              customInput.required = true;

              nameFields.classList.replace('md:col-span-2', 'md:col-span-12'); 

          } else {

              customWrapper.classList.add('hidden');

              customInput.required = false;

              customInput.value = '';

          }

      }


      function handlePdpaChange() {

          const pdpaVal = document.querySelector('input[name="pdpa"]:checked').value;

          const submitBtn = document.getElementById('submitBtn');

          const declineMsg = document.getElementById('decline-msg');


          if (pdpaVal === 'accept') {

              declineMsg.classList.add('hidden');

              submitBtn.disabled = false;

              submitBtn.classList.replace('bg-gray-300', 'bg-cra-blue');

              submitBtn.classList.replace('text-gray-500', 'text-white');

              submitBtn.classList.replace('cursor-not-allowed', 'hover:bg-cra-lightBlue');

              submitBtn.classList.add('shadow-lg', 'shadow-blue-500/30');

          } else {

              declineMsg.classList.remove('hidden');

              submitBtn.disabled = true;

              submitBtn.classList.replace('bg-cra-blue', 'bg-gray-300');

              submitBtn.classList.replace('text-white', 'text-gray-500');

              submitBtn.classList.add('cursor-not-allowed');

              submitBtn.classList.remove('shadow-lg', 'shadow-blue-500/30', 'hover:bg-cra-lightBlue');

          }

      }


      function validateAndSubmit() {

          const errorBox = document.getElementById('error-msg');

          const errorText = errorBox.querySelector('span');

          errorBox.classList.add('hidden');


          const craId = document.getElementById('craId').value;

          const prefix = document.getElementById('prefix').value;

          const customPrefix = document.getElementById('customPrefix').value;

          const firstName = document.getElementById('firstName').value.trim();

          const lastName = document.getElementById('lastName').value.trim();

          const affiliation = document.getElementById('affiliation').value;

          const username = document.getElementById('emailUsername').value.trim();

          const domain = document.getElementById('emailDomain').value;

          

          if (!craId || !prefix || !firstName || !lastName || !affiliation || !username) {

              errorText.innerText = "กรุณากรอกข้อมูลในช่องที่มีเครื่องหมาย * ให้ครบถ้วน";

              errorBox.classList.remove('hidden');

              return;

          }


          if (prefix === 'อื่นๆ' && !customPrefix) {

              errorText.innerText = "กรุณาระบุคำนำหน้าชื่อของคุณ";

              errorBox.classList.remove('hidden');

              return;

          }


          const userData = {

              userType: document.getElementById('type-student').checked ? 'นักศึกษา' : 'บุคลากร',

              craId: craId,

              name: (prefix === 'อื่นๆ' ? customPrefix : prefix) + firstName + ' ' + lastName, 

              department: affiliation,

              email: `${username}@${domain}`

          };


          // บันทึกตัวแปรให้ตรงกับที่หน้ารีวิวต้องการดึงไปแสดงผล

          localStorage.setItem('cra_user_info', JSON.stringify(userData));


          // เคลียร์ตะกร้าสินค้าเก่าทิ้ง เพื่อให้คนใหม่เริ่มคัดเลือกจาก 0

          localStorage.removeItem('cra_selected_mags');

          localStorage.removeItem('cra_cart');


          const btn = document.getElementById('submitBtn');

          btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> กำลังไปหน้าถัดไป...';

          btn.disabled = true;


          setTimeout(() => {

              goToPage('select'); 

          }, 500);

      }

  </script>
