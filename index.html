  <script>

      const API_URL = "https://script.google.com/macros/s/AKfycbxWqJMu2kgMFqs_7x-RfjUyRuO1Y86ZQJQUeCEqoCv4NKtBDejG3XN7n0z0dLnGymBbAw/exec";


      // ฟังก์ชันสำหรับเปลี่ยนหน้า (ทำงานบน GitHub Pages)

      function goToPage(pageName) {

          const btn = document.querySelector('button[onclick^="goToPage"]');

          if(btn) {

              btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> กำลังโหลด...';

              btn.disabled = true;

          }

          

          // ลิงก์ตรงไปที่ไฟล์ HTML บน GitHub Pages

          window.location.href = pageName + ".html";

      }


      // Set fixed deadline for visual purposes

      const updateCountdown = () => {

          const endDate = new Date('2026-10-15T23:59:59').getTime();

          const now = new Date().getTime();

          const distance = endDate - now;


          if (distance > 0) {

              const days = Math.floor(distance / (1000 * 60 * 60 * 24));

              const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

              const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));

              const seconds = Math.floor((distance % (1000 * 60)) / 1000);


              document.getElementById('cd-days').innerText = String(days).padStart(2, '0');

              document.getElementById('cd-hours').innerText = String(hours).padStart(2, '0');

              document.getElementById('cd-mins').innerText = String(minutes).padStart(2, '0');

              document.getElementById('cd-secs').innerText = String(seconds).padStart(2, '0');

          } else {

              document.getElementById('cd-days').innerText = "00";

              document.getElementById('cd-hours').innerText = "00";

              document.getElementById('cd-mins').innerText = "00";

              document.getElementById('cd-secs').innerText = "00";

          }

      };

      

      setInterval(updateCountdown, 1000); // Update every 1 second

      updateCountdown(); 


      document.addEventListener("DOMContentLoaded", () => {

          // ดึงข้อมูลนิตยสารแนะนำจาก API

          fetch(API_URL + "?action=getMagazines")

            .then(res => res.json())

            .then(magazines => {

                if (!magazines || magazines.length === 0) return;

                

                // ค้นหากล่องหน้าปก Magazine ทั้งหมดบนหน้าจอ

                const covers = document.querySelectorAll('.mag-cover');

                

                covers.forEach((cover, index) => {

                    // ถ้ามีข้อมูลแมกกาซีนสำหรับปกนี้

                    if (magazines[index]) {

                        const mag = magazines[index];

                        

                        // หา key ของ Title และ Category (รองรับชื่อคอลัมน์หลายแบบ)

                        const title = mag.title || mag['ชื่อเรื่อง'] || mag['ชื่อหนังสือ'] || 'e-Magazine';

                        const category = mag.category || mag['หมวดหมู่'] || 'Featured';

                        

                        // หา Element ที่ใช้แสดงข้อความบนปก

                        const titleEl = cover.querySelector('h3');

                        const catEl = cover.querySelector('.uppercase'); 

                        

                        if (titleEl) {

                            titleEl.innerText = title;

                            // ปรับขนาดฟอนต์ให้เล็กลง เพื่อรองรับชื่อเรื่องยาวๆ ของจริง

                            titleEl.classList.remove('text-3xl', 'text-4xl', 'text-5xl', 'font-black');

                            titleEl.classList.add('text-xl', 'font-bold', 'leading-snug');

                            // ตัดคำอัตโนมัติหากยาวไป

                            titleEl.style.display = '-webkit-box';

                            titleEl.style.webkitLineClamp = '3';

                            titleEl.style.webkitBoxOrient = 'vertical';

                            titleEl.style.overflow = 'hidden';

                        }

                        

                        if (catEl) {

                            catEl.innerText = category;

                        }

                    }

                });

            })

            .catch(err => console.error("Failed to load featured magazines", err));

      });
