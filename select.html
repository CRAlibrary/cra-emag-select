  <script>

      const API_URL = "https://script.google.com/macros/s/AKfycbxWqJMu2kgMFqs_7x-RfjUyRuO1Y86ZQJQUeCEqoCv4NKtBDejG3XN7n0z0dLnGymBbAw/exec";


      // --- State Variables ---

      let allMagazines = []; 

      let filteredMagazines = []; 

      let selectedItems = []; 

      

      let currentPage = 1;

      const itemsPerPage = 24;

      let activeCategory = 'all';


      // --- Routing Function ---

      function goToPage(pageName) {

          if (pageName === 'review' && selectedItems.length === 0) {

              alert("กรุณาเลือก e-Magazine อย่างน้อย 1 รายการก่อนครับ");

              return;

          }


          const btn = document.querySelector(`button[onclick="goToPage('${pageName}')"]`);

          if(btn) btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> กำลังไป...';

          

          // เซฟข้อมูลตะกร้า (เซฟไว้ 2 ชื่อเลย เพื่อให้รองรับกับหน้ารีวิวทุกเวอร์ชัน)

          localStorage.setItem('cra_cart', JSON.stringify(selectedItems));

          localStorage.setItem('cra_selected_mags', JSON.stringify(selectedItems));

          

          window.location.href = pageName + '.html';

      }


      // --- Initialization ---

      document.addEventListener("DOMContentLoaded", () => {

          // Load saved cart from session

          try {

              const savedCart = localStorage.getItem('cra_cart') || localStorage.getItem('cra_selected_mags');

              if(savedCart) {

                  selectedItems = JSON.parse(savedCart);

                  if (!Array.isArray(selectedItems)) selectedItems = [];

              }

          } catch (e) {

              selectedItems = [];

              localStorage.removeItem('cra_cart');

              localStorage.removeItem('cra_selected_mags');

          }

          updateStickyBar();


          // Fetch Data

          fetch(API_URL + "?action=getMagazines")

            .then(res => res.json())

            .then(data => handleDataLoad(data))

            .catch(err => {

               console.error(err); 

               document.getElementById('loadingState').innerHTML = '<div class="text-center py-10"><i class="fas fa-exclamation-triangle text-red-500 text-4xl mb-4"></i><p class="text-red-500 font-bold">เกิดข้อผิดพลาดในการดึงข้อมูล</p><p class="text-gray-500 text-sm mt-2">โปรดตรวจสอบ API URL</p></div>'; 

            });


          // Event Listeners

          document.getElementById('searchInput').addEventListener('input', (e) => applyFilters());

          document.getElementById('sortSelect').addEventListener('change', () => applyFilters());

          document.getElementById('loadMoreBtn').addEventListener('click', () => {

              currentPage++;

              renderGrid(true); 

          });

      });


      // --- Data Processing ---

      function handleDataLoad(data) {

          document.getElementById('loadingState').classList.add('hidden');

          document.getElementById('magazineGrid').classList.remove('hidden');

          

          if (data.error) {

              document.getElementById('magazineGrid').innerHTML = `<div class="col-span-full text-center text-red-500 py-10">Error: ${data.error}</div>`;

              return;

          }


          allMagazines = data || [];

          filteredMagazines = [...allMagazines];

          

          renderSidebar();

          applyFilters();

      }


      function getCategoryColorClass(cat) {

          if (!cat) return 'cat-default';

          const c = cat.toLowerCase();

          if(c.includes('industry') || c.includes('trade')) return 'cat-industry';

          if(c.includes('art') || c.includes('design')) return 'cat-art';

          if(c.includes('hobbies') || c.includes('diy') || c.includes('interest')) return 'cat-hobbies';

          if(c.includes('family') || c.includes('kid') || c.includes('teen')) return 'cat-family';

          if(c.includes('travel') || c.includes('region')) return 'cat-travel';

          if(c.includes('consumer') || c.includes('sport') || c.includes('health')) return 'cat-consumer';

          if(c.includes('business') || c.includes('finance') || c.includes('manage')) return 'cat-business';

          return 'cat-default';

      }


      // --- Rendering ---

      function renderSidebar() {

          const catCounts = allMagazines.reduce((acc, mag) => {

              const category = mag.category || 'Uncategorized';

              acc[category] = (acc[category] || 0) + 1;

              return acc;

          }, {});


          const sortedCats = Object.keys(catCounts).sort();

          

          let html = `

              <button onclick="selectCategory('all')" id="btn-cat-all" class="cat-btn w-full text-left px-3 py-2 rounded-lg text-sm font-medium transition-colors bg-blue-50 text-cra-blue">

                  ทั้งหมด <span class="float-right bg-blue-100 text-cra-blue text-[10px] px-2 py-0.5 rounded-full">${allMagazines.length}</span>

              </button>

          `;


          sortedCats.forEach(cat => {

              html += `

                  <button onclick="selectCategory('${cat}')" id="btn-cat-${cssSafe(cat)}" class="cat-btn w-full text-left px-3 py-2 rounded-lg text-sm font-medium text-gray-600 hover:bg-gray-50 transition-colors mt-1">

                      ${cat} <span class="float-right bg-gray-100 text-gray-500 text-[10px] px-2 py-0.5 rounded-full">${catCounts[cat]}</span>

                  </button>

              `;

          });


          document.getElementById('category-list').innerHTML = html;

      }


      function cssSafe(str) {

          return str.replace(/[^a-zA-Z0-9-]/g, '-');

      }


      function renderGrid(append = false) {

          const grid = document.getElementById('magazineGrid');

          if (!append) grid.innerHTML = ''; 


          if (filteredMagazines.length === 0) {

              grid.classList.add('hidden');

              document.getElementById('emptyState').classList.remove('hidden');

              document.getElementById('loadMoreBtn').classList.add('hidden');

              return;

          }


          grid.classList.remove('hidden');

          document.getElementById('emptyState').classList.add('hidden');


          const startIndex = (currentPage - 1) * itemsPerPage;

          const endIndex = startIndex + itemsPerPage;

          const pageData = filteredMagazines.slice(startIndex, endIndex);


          pageData.forEach(mag => {

              const isSelected = selectedItems.some(item => item.id === mag.id);

              const title = mag.title || 'Untitled';

              const firstLetter = title.replace(/[^a-zA-Z0-9ก-๙]/g, '').charAt(0).toUpperCase() || 'E';

              const colorClass = getCategoryColorClass(mag.category);

              

              const card = document.createElement('div');

              card.className = `mag-card group flex flex-col bg-white rounded-2xl p-3 border border-gray-100 shadow-sm hover:shadow-md transition-all ${isSelected ? 'selected' : ''}`;

              card.dataset.id = mag.id;

              

              card.innerHTML = `

                  <div class="book-cover ${colorClass} mb-3 cursor-pointer" onclick="toggleSelection('${mag.id}')">

                      <div class="cover-content">

                          <div class="text-[9px] uppercase tracking-wider opacity-80 mb-1 leading-tight line-clamp-2">${mag.category || 'Uncategorized'}</div>

                          <h3 class="font-serif font-bold text-sm leading-tight line-clamp-4" title="${title}">${title}</h3>

                      </div>

                      <div class="cover-letter">${firstLetter}</div>

                  </div>

                  <div class="mt-auto flex flex-col px-1">

                      <div class="text-sm font-bold text-gray-800 mb-3">฿${(mag.price || 0).toLocaleString()}</div>

                      <button onclick="toggleSelection('${mag.id}')" class="select-btn w-full py-2 rounded-xl text-xs font-bold border border-gray-200 text-gray-600 hover:bg-gray-50 flex items-center justify-center gap-1">

                          ${isSelected ? '<i class="fas fa-check"></i> เลือกแล้ว' : 'เลือกเล่มนี้'}

                      </button>

                  </div>

              `;

              grid.appendChild(card);

          });


          document.getElementById('resultCount').innerText = `${filteredMagazines.length} รายการ`;

          if (endIndex >= filteredMagazines.length) {

              document.getElementById('loadMoreBtn').classList.add('hidden');

          } else {

              document.getElementById('loadMoreBtn').classList.remove('hidden');

          }

      }


      // --- Interaction Logic ---

      function selectCategory(cat) {

          activeCategory = cat;

          

          document.querySelectorAll('.cat-btn').forEach(btn => {

              btn.classList.remove('bg-blue-50', 'text-cra-blue');

              btn.classList.add('text-gray-600');

              const badge = btn.querySelector('span');

              badge.classList.replace('bg-blue-100', 'bg-gray-100');

              badge.classList.replace('text-cra-blue', 'text-gray-500');

          });


          const activeBtn = document.getElementById(cat === 'all' ? 'btn-cat-all' : `btn-cat-${cssSafe(cat)}`);

          if (activeBtn) {

              activeBtn.classList.remove('text-gray-600');

              activeBtn.classList.add('bg-blue-50', 'text-cra-blue');

              const badge = activeBtn.querySelector('span');

              badge.classList.replace('bg-gray-100', 'bg-blue-100');

              badge.classList.replace('text-gray-500', 'text-cra-blue');

          }


          document.getElementById('currentCategoryTitle').innerText = cat === 'all' ? 'นิตยสารทั้งหมด' : cat;

          applyFilters();

      }


      function applyFilters() {

          const searchTerm = document.getElementById('searchInput').value.toLowerCase();

          const sortVal = document.getElementById('sortSelect').value;


          filteredMagazines = allMagazines.filter(mag => {

              const magCat = mag.category || 'Uncategorized';

              const matchCat = activeCategory === 'all' || magCat === activeCategory;

              const title = mag.title || '';

              const matchSearch = title.toLowerCase().includes(searchTerm);

              return matchCat && matchSearch;

          });


          filteredMagazines.sort((a, b) => {

              const titleA = a.title || '';

              const titleB = b.title || '';

              const priceA = a.price || 0;

              const priceB = b.price || 0;


              if (sortVal === 'az') return titleA.localeCompare(titleB);

              if (sortVal === 'za') return titleB.localeCompare(titleA);

              if (sortVal === 'priceAsc') return priceA - priceB;

              if (sortVal === 'priceDesc') return priceB - priceA;

              return 0;

          });


          currentPage = 1;

          renderGrid();

      }


      function clearFilters() {

          document.getElementById('searchInput').value = '';

          selectCategory('all');

      }


      function toggleSelection(id) {

          const mag = allMagazines.find(m => m.id === id);

          if (!mag) return;


          const index = selectedItems.findIndex(item => item.id === id);

          

          if (index > -1) {

              selectedItems.splice(index, 1); 

          } else {

              selectedItems.push(mag); 

          }

          

          // Save cart to local storage immediately

          localStorage.setItem('cra_cart', JSON.stringify(selectedItems));

          localStorage.setItem('cra_selected_mags', JSON.stringify(selectedItems));

          

          const card = document.querySelector(`.mag-card[data-id="${id}"]`);

          if (card) {

              const isNowSelected = index === -1; 

              card.classList.toggle('selected', isNowSelected);

              const btn = card.querySelector('.select-btn');

              if (isNowSelected) {

                  btn.innerHTML = '<i class="fas fa-check"></i> เลือกแล้ว';

                  card.style.transform = 'scale(0.98)';

                  setTimeout(() => card.style.transform = '', 150);

              } else {

                  btn.innerHTML = 'เลือกเล่มนี้';

              }

          }


          updateStickyBar();

      }


      function updateStickyBar() {

          const bar = document.getElementById('stickyBar');

          const countEl = document.getElementById('cartCount');

          const badgeEl = document.getElementById('cartBadge');

          const totalEl = document.getElementById('cartTotal');

          

          const count = selectedItems.length;

          const total = selectedItems.reduce((sum, item) => sum + (item.price || 0), 0);

          

          countEl.innerText = count;

          badgeEl.innerText = count;

          totalEl.innerText = total.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});


          if (count > 0) {

              bar.classList.add('visible');

          } else {

              bar.classList.remove('visible');

          }

      }
