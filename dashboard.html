      function fetchData() {

          const loader = document.getElementById('loader');

          const content = document.getElementById('dashboardContent');

          const refreshBtn = document.getElementById('refreshBtn');

          const refreshIcon = refreshBtn.querySelector('i');

            

          loader.classList.remove('hidden');

          loader.classList.add('flex');

          content.classList.add('hidden');

          refreshIcon.classList.add('fa-spin');


          fetch(API_URL + "?action=getDashboardMetrics")

            .then(res => res.json())

            .then(response => {

                refreshIcon.classList.remove('fa-spin');

                loader.classList.remove('flex');

                loader.classList.add('hidden');

                    

                if (response.success && response.metrics) {

                    const m = response.metrics;

                        

                    document.getElementById('m-users').innerText = m.uniqueUsers.toLocaleString();

                    document.getElementById('m-submissions').innerText = m.totalSubmissions.toLocaleString() + " ครั้ง";

                    document.getElementById('m-items').innerText = m.totalItemsSelected.toLocaleString();

                    document.getElementById('m-budget').innerText = '฿' + m.totalBudget.toLocaleString('th-TH', {minimumFractionDigits: 2});

                    document.getElementById('m-avg').innerText = '฿' + m.avgBudgetPerUser.toLocaleString('th-TH', {minimumFractionDigits: 2});

                        

                    renderCharts(m.charts);


                    allTopMagazines = m.topMagazines;

                    magsToShow = 10;

                    renderTopTable();


                    content.classList.remove('hidden');

                } else {

                    alert("เกิดข้อผิดพลาด: " + (response.message || "ไม่สามารถดึงข้อมูลได้"));

                }

            })

            .catch(error => {

                refreshIcon.classList.remove('fa-spin');

                loader.classList.remove('flex');

                loader.classList.add('hidden');

                alert("Error connection: " + error.message);

            });

      }


      function renderCharts(chartData) {

          Chart.register(ChartDataLabels);

          Chart.defaults.set('plugins.datalabels', { display: false });


          ['userTypeChart', 'categoryChart', 'deptChart'].forEach(id => {

              if (chartInstances[id]) chartInstances[id].destroy();

          });


          // 1. User Type Doughnut Chart

          const ctxUser = document.getElementById('userTypeChart').getContext('2d');

          chartInstances['userTypeChart'] = new Chart(ctxUser, {

              type: 'doughnut',

              data: {

                  labels: chartData.userTypes.map(d => d.label),

                  datasets: [{

                      data: chartData.userTypes.map(d => d.value),

                      backgroundColor: ['#2563eb', '#f97316', '#10b981', '#f59e0b', '#8b5cf6'],

                      borderWidth: 2,

                      borderColor: '#ffffff',

                      hoverOffset: 12

                  }]

              },

              options: {

                  responsive: true, maintainAspectRatio: false,

                  plugins: { 

                      legend: { position: 'bottom' },

                      datalabels: {

                          display: true,

                          color: '#fff',

                          font: { weight: 'bold', size: 14 },

                          formatter: (value, ctx) => {

                              let sum = ctx.chart._metasets[ctx.datasetIndex].total;

                              if (sum === 0) return '';

                              return (value * 100 / sum).toFixed(1) + "%";

                          }

                      },

                      tooltip: {

                          callbacks: {

                              label: function(context) {

                                  let label = context.label || '';

                                  let value = context.raw;

                                  let sum = context.chart._metasets[context.datasetIndex].total;

                                  let percentage = sum > 0 ? (value * 100 / sum).toFixed(1) + "%" : "0%";

                                  return ` ${label}: ${value} คน (${percentage})`;

                              }

                          }

                      }

                  },

                  cutout: '65%'

              }

          });


          // 2. Categories Bar Chart

          const ctxCat = document.getElementById('categoryChart').getContext('2d');

          chartInstances['categoryChart'] = new Chart(ctxCat, {

              type: 'bar',

              data: {

                  labels: chartData.categories.map(d => d.label),

                  datasets: [{

                      label: 'จำนวนที่ถูกเลือก (ครั้ง)',

                      data: chartData.categories.map(d => d.value),

                      backgroundColor: '#f97316',

                      borderRadius: 6,

                      hoverBackgroundColor: '#ea580c'

                  }]

              },

              options: {

                  responsive: true, maintainAspectRatio: false,

                  plugins: { 

                      legend: { display: false },

                      datalabels: {

                          display: true,

                          color: '#f97316',

                          anchor: 'end',

                          align: 'top',

                          font: { weight: 'bold' }

                      },

                      tooltip: { callbacks: { label: (ctx) => ` ถูกเลือก: ${ctx.raw} ครั้ง` } }

                  },

                  scales: { y: { beginAtZero: true, grid: { borderDash: [2, 4] }, suggestedMax: Math.max(...chartData.categories.map(d => d.value)) + 2 } }

              }

          });

          const ctxDept = document.getElementById('deptChart').getContext('2d');

          chartInstances['deptChart'] = new Chart(ctxDept, {

              type: 'bar',

              data: {

                  labels: chartData.departments.map(d => d.label.length > 15 ? d.label.substring(0, 15) + '...' : d.label),

                  datasets: [{

                      label: 'ผู้ใช้',

                      data: chartData.departments.map(d => d.value),

                      backgroundColor: '#10b981',

                      borderRadius: 6,

                      hoverBackgroundColor: '#059669'

                  }]

              },

              options: {

                  indexAxis: 'y',

                  responsive: true, maintainAspectRatio: false,

                  plugins: { 

                      legend: { display: false },

                      datalabels: {

      function renderTopTable() {

          const tbody = document.getElementById('topMagazinesList');

          const emptyState = document.getElementById('emptyTableState');

          const loadMoreBtn = document.getElementById('loadMoreMagsBtn');

            

          if (allTopMagazines && allTopMagazines.length > 0) {

              emptyState.classList.add('hidden');

                

              let html = '';

              const displayMags = allTopMagazines.slice(0, magsToShow);


              displayMags.forEach((mag, index) => {

                  let rankClass = 'text-gray-500';

                  let rankIcon = `${index + 1}`;

                    

                  if (index === 0) { rankClass = 'text-yellow-500 font-bold text-base'; }

                  else if (index === 1) { rankClass = 'text-gray-400 font-bold text-base'; }

                  else if (index === 2) { rankClass = 'text-amber-700 font-bold text-base'; }


                  html += `

                      <tr class="hover:bg-blue-50 transition-colors cursor-default">

                          <td class="px-6 py-4 whitespace-nowrap text-center text-sm ${rankClass}">${rankIcon}</td>

                          <td class="px-6 py-4 text-sm font-medium text-gray-800 line-clamp-1" title="${mag.title}">${mag.title}</td>

                          <td class="px-6 py-4 whitespace-nowrap text-center">

                              <span class="px-2.5 py-1 inline-flex text-xs font-bold rounded-full bg-blue-100 text-blue-800 shadow-sm">

                                  ${mag.count}

                              </span>

                          </td>

                          <td class="px-6 py-4 whitespace-nowrap text-right text-sm font-bold text-emerald-600">

                                ฿${mag.price.toLocaleString('th-TH', {minimumFractionDigits: 2})}

                          </td>

                      </tr>

                  `;

              });

              tbody.innerHTML = html;


              if (loadMoreBtn) {

                  if (magsToShow < allTopMagazines.length) {

                      loadMoreBtn.classList.remove('hidden');

                  } else {

                      loadMoreBtn.classList.add('hidden');

                  }

              }

          } else {

              tbody.innerHTML = '';

              emptyState.classList.remove('hidden');

              if (loadMoreBtn) loadMoreBtn.classList.add('hidden');

          }

      }


      function loadMoreMagazines() {

          magsToShow += 10;

          renderTopTable();

      }
